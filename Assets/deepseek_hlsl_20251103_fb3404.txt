Shader "Sprites/Outline"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1,1,1,1)
        _OutlineColor ("Outline Color", Color) = (1,1,1,1)
        _OutlineWidth ("Outline Width", Float) = 0.01
    }

    SubShader
    {
        Tags
        {
            "Queue"="Transparent"
            "RenderType"="Transparent"
        }

        Blend SrcAlpha OneMinusSrcAlpha

        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "UnityCG.cginc"

            struct appdata
            {
                float4 vertex : POSITION;
                float2 uv : TEXCOORD0;
            };

            struct v2f
            {
                float2 uv : TEXCOORD0;
                float4 vertex : SV_POSITION;
            };

            sampler2D _MainTex;
            float4 _MainTex_ST;
            fixed4 _Color;
            fixed4 _OutlineColor;
            float _OutlineWidth;

            v2f vert (appdata v)
            {
                v2f o;
                o.vertex = UnityObjectToClipPos(v.vertex);
                o.uv = TRANSFORM_TEX(v.uv, _MainTex);
                return o;
            }

            fixed4 frag (v2f i) : SV_Target
            {
                fixed4 col = tex2D(_MainTex, i.uv);
                
                // Проверяем границы
                if (col.a < 0.5)
                {
                    fixed4 up = tex2D(_MainTex, i.uv + float2(0, _OutlineWidth));
                    fixed4 down = tex2D(_MainTex, i.uv - float2(0, _OutlineWidth));
                    fixed4 left = tex2D(_MainTex, i.uv - float2(_OutlineWidth, 0));
                    fixed4 right = tex2D(_MainTex, i.uv + float2(_OutlineWidth, 0));
                    
                    if (up.a > 0.5 || down.a > 0.5 || left.a > 0.5 || right.a > 0.5)
                    {
                        return _OutlineColor;
                    }
                }
                
                return col * _Color;
            }
            ENDCG
        }
    }
}
